  <script>
    function convertTemp() {
      const input = document.getElementById("tempInput").value.trim();
      const unit = document.getElementById("unitSelect").value;
      const resultText = document.getElementById("resultText");
      const resultBox = document.getElementById("resultBox");

      if (isNaN(input) || input === "") {
        resultText.innerText = "⚠️ Please enter a valid number.";
        resultText.classList.add("error");
        resultBox.style.display = "block";
        return;
      }

      const temp = parseFloat(input);
      let result = "";

      switch(unit) {
        case "C":
          result = `${((temp * 9/5) + 32).toFixed(2)} °F\n${(temp + 273.15).toFixed(2)} K`;
          break;
        case "F":
          const c = (temp - 32) * 5/9;
          result = `${c.toFixed(2)} °C\n${(c + 273.15).toFixed(2)} K`;
          break;
        case "K":
          const c2 = temp - 273.15;
          result = `${c2.toFixed(2)} °C\n${((c2 * 9/5) + 32).toFixed(2)} °F`;
          break;
      }

      resultText.innerText = result;
      resultText.classList.remove("error");
      resultBox.style.display = "block";
    }
  </script>
